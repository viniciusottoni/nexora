using Awaken.Domain.Entities.Quests;
using Awaken.Domain.Repositories;
using Microsoft.EntityFrameworkCore;

namespace Awaken.Infrastructure.Persistence.Repositories;

public class QuestLogRepository(AwakenDbContext context) : IQuestLogRepository
{
    public async Task<QuestLog?> GetByIdAsync(Guid id, CancellationToken cancellationToken = default) =>
        await context.QuestLogs.FirstOrDefaultAsync(l => l.Id == id, cancellationToken);

    public async Task<IEnumerable<QuestLog>> GetAllAsync(CancellationToken cancellationToken = default) =>
        await context.QuestLogs.ToListAsync(cancellationToken);

    public async Task AddAsync(QuestLog entity, CancellationToken cancellationToken = default) =>
        await context.QuestLogs.AddAsync(entity, cancellationToken);

    public void Update(QuestLog entity) => context.QuestLogs.Update(entity);

    public void Remove(QuestLog entity) => context.QuestLogs.Remove(entity);

    public async Task<QuestLog?> GetByQuestIdAsync(Guid questId, CancellationToken cancellationToken = default) =>
        await context.QuestLogs.FirstOrDefaultAsync(l => l.QuestId == questId, cancellationToken);

    public async Task<IReadOnlyList<QuestLog>> GetRecentByUserIdAsync(
        Guid userId, int limit, CancellationToken cancellationToken = default) =>
        await context.QuestLogs
            .Where(l => l.UserId == userId)
            .OrderByDescending(l => l.CompletedAtUtc)
            .Take(limit)
            .ToListAsync(cancellationToken);
}

```csharp
using Awaken.Application.Common.Interfaces;
using Awaken.Contracts.BattleLog;
using Awaken.Domain.Repositories;
using MediatR;

namespace Awaken.Application.BattleLog.Queries.GetRecentBattleLog;

/// US-081: retorna os logs de quests concluidas do usuario autenticado,
/// ordenados do mais recente ao mais antigo. Backend e autoridade (ADR-009).
public class GetRecentBattleLogQueryHandler(
    IQuestLogRepository questLogRepository,
    ICurrentUserService currentUserService) : IRequestHandler<GetRecentBattleLogQuery, BattleLogResponse>
{
    public async Task<BattleLogResponse> Handle(
        GetRecentBattleLogQuery request,
        CancellationToken cancellationToken)
    {
        var userId = currentUserService.UserId;

        var logs = await questLogRepository.GetRecentByUserIdAsync(
            userId, request.Limit, cancellationToken);

        var items = logs
            .Select(l => new BattleLogItemResponse(
                l.Id,
                l.QuestId,
                l.QuestType,
                l.XpEarned,
                l.ItemsEarned,
                l.CompletedAtUtc))
            .ToList();

        return new BattleLogResponse(items);
    }
}
```

    required this.itemsEarned,

  final List<String> itemsEarned;
  final DateTime completedAt;

      questLogId: json['questLogId'] as String,
      questId: json['questId'] as String,
      questType: json['questType'] as String,
      xpEarned: json['xpEarned'] as int,
      itemsEarned: (json['itemsEarned'] as List<dynamic>)
          .map((e) => e as String)
          .toList(),
      completedAt: DateTime.parse(json['completedAt'] as String),
    );
  }
}

  const BattleLogResponseDto({required this.items});

    return BattleLogResponseDto(
      items: (json['items'] as List<dynamic>)
          .map((e) => BattleLogItemDto.fromJson(e as Map<String, dynamic>))
          .toList(),
    );

import '../dtos/battle_log_response_dto.dart';

class BattleLogRemoteDataSource {
  const BattleLogRemoteDataSource(this._dio);
  final Dio _dio;

  /// US-081: busca os logs de quests concluidas recentemente.
  Future<BattleLogResponseDto> getRecentBattleLog() async {
    try {
      final response = await _dio.get('/api/hunter/battle-log/recent');
      return BattleLogResponseDto.fromJson(response.data as Map<String, dynamic>);
    } on DioException catch (e) {
      throw _mapError(e);
    }
  }

  AppError _mapError(DioException e) {
    if (e.response?.statusCode == 403) return const AccessBlockedError();
    if (e.type == DioExceptionType.connectionTimeout ||
        e.type == DioExceptionType.sendTimeout ||
        e.type == DioExceptionType.receiveTimeout ||
        e.type == DioExceptionType.connectionError) {
      return const NetworkError();
    }
    return const UnexpectedError();
  }
}

import '../../../../core/analytics/analytics_provider.dart';

/// US-081: carrega o log de batalha do usuario autenticado.
class BattleLogController extends Notifier<BattleLogState> {
  @override
  BattleLogState build() => const BattleLogLoading();

  Future<void> load() async {
    state = const BattleLogLoading();

    final repository = ref.read(battleLogRepositoryProvider);
    final analytics = ref.read(analyticsServiceProvider);

    try {
      final items = await repository.getRecentBattleLog();

      if (items.isEmpty) {
        state = const BattleLogEmpty();
        await analytics.logEvent('battle_log_empty_viewed');
      } else {
        state = BattleLogLoaded(items);
        await analytics.logEvent('battle_log_viewed');
      }
    } on AccessBlockedError {
      final session = ref.read(currentSessionStateProvider);
      final accessStatus = session?.accessStatus ?? AccessStatus.trialExpired;
      state = BattleLogAccessBlocked(accessStatus);
    } on AppError catch (e) {
      state = BattleLogError(e);
    } catch (_) {
      state = const BattleLogError(UnexpectedError());
    }
  }
}

final battleLogControllerProvider =
    NotifierProvider<BattleLogController, BattleLogState>(
        BattleLogController.new);
